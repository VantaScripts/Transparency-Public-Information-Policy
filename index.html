-- =============================== -- VANTA DEMO SCRIPT (EXECUTOR READY) -- =============================== -- Ensure Fluent is loaded local Fluent = _G.Fluent or error("Fluent library not found! Make sure Fluent is loaded.") -- Paste VantaLibrary here or require it if stored externally local VantaLibrary = {} do local lib = {} lib.__index = lib function lib.new() local self = setmetatable({}, lib) self.Window = nil self.Tabs = {} self.Options = Fluent.Options return self end function lib:CreateWindow(title, subtitle, size, acrylic, theme, minimizeKey) self.Window = Fluent:CreateWindow({ Title = title or "Vanta UI", SubTitle = subtitle or "", Size = size or UDim2.fromOffset(580, 460), Acrylic = acrylic ~= false, Theme = theme or "Dark", MinimizeKey = minimizeKey or Enum.KeyCode.LeftControl }) return self.Window end function lib:AddTab(name, icon) local tab = self.Window:AddTab({Title = name, Icon = icon or ""}) self.Tabs[name] = tab return tab end function lib:AddButton(tab, title, description, callback) return tab:AddButton({Title = title, Description = description, Callback = callback}) end function lib:AddToggle(tab, key, title, default, onChanged) local toggle = tab:AddToggle(key, {Title = title, Default = default}) if onChanged then toggle:OnChanged(function() onChanged(self.Options[key].Value) end) end return toggle end function lib:AddSlider(tab, key, title, description, default, min, max, rounding, callback) local slider = tab:AddSlider(key, { Title = title, Description = description, Default = default, Min = min, Max = max, Rounding = rounding, Callback = callback }) return slider end function lib:AddDropdown(tab, key, title, values, multi, default, onChanged) local dropdown = tab:AddDropdown(key, {Title = title, Values = values, Multi = multi, Default = default}) if onChanged then dropdown:OnChanged(onChanged) end return dropdown end function lib:AddColorpicker(tab, key, title, default, transparency, onChanged) local cp = tab:AddColorpicker(key, {Title = title, Default = default or Color3.fromRGB(255,255,255), Transparency = transparency or nil}) if onChanged then cp:OnChanged(onChanged) end return cp end function lib:AddKeybind(tab, key, title, mode, default, callback, changedCallback) local kb = tab:AddKeybind(key, {Title = title, Mode = mode or "Toggle", Default = default or "LeftControl", Callback = callback, ChangedCallback = changedCallback}) return kb end function lib:AddInput(tab, key, title, default, placeholder, numeric, finished, callback) local input = tab:AddInput(key, {Title = title, Default = default, Placeholder = placeholder, Numeric = numeric or false, Finished = finished or false, Callback = callback}) return input end function lib:Notify(title, content, subcontent, duration) Fluent:Notify({Title = title, Content = content, SubContent = subcontent, Duration = duration}) end function lib:Dialog(title, content, buttons) self.Window:Dialog({Title = title, Content = content, Buttons = buttons}) end VantaLibrary = lib end -- =============================== -- CREATE UI -- =============================== local UI = VantaLibrary.new() -- Window UI:CreateWindow("Vanta UI", "", UDim2.fromOffset(580, 460)) -- Tabs local mainTab = UI:AddTab("Main") local settingsTab = UI:AddTab("Settings", "settings") -- Button UI:AddButton(mainTab, "Test Button", "Click me!", function() UI:Dialog("Button Clicked", "You clicked the button!", {{Title="OK", Callback=function() print("OK clicked") end}}) end) -- Toggle UI:AddToggle(mainTab, "DemoToggle", "Enable Feature", false, function(val) print("Toggle:", val) end) -- Slider UI:AddSlider(mainTab, "DemoSlider", "Adjust Value", "Slide me", 2, 0, 5, 1, function(val) print("Slider:", val) end) -- Dropdown UI:AddDropdown(mainTab, "DemoDropdown", "Select Option", {"One","Two","Three"}, false, "One", function(val) print("Dropdown selected:", val) end) -- Multi-dropdown UI:AddDropdown(mainTab, "DemoMulti", "Multi Select", {"One","Two","Three"}, true, {"One"}, function(val) local selected = {} for k,v in pairs(val) do if v then table.insert(selected,k) end end print("Multi dropdown:", table.concat(selected,", ")) end) -- Colorpicker local cp = UI:AddColorpicker(mainTab, "DemoColor", "Pick Color", Color3.fromRGB(100,200,255), 0, function() print("Color:", cp.Value) end) -- Keybind UI:AddKeybind(mainTab, "DemoKey", "Press Key", "Toggle", "MB2", function(val) print("Keybind pressed:", val) end, function(new) print("Keybind changed:", new) end) -- Input UI:AddInput(mainTab, "DemoInput", "Type Here", "Default", "Placeholder", false, false, function(val) print("Input value:", val) end) -- Notification on load UI:Notify("Vanta UI", "All controls loaded!", nil, 5)
