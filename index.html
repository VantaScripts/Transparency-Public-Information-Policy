--[[ VantaLibrary Complete Roblox executor UI library using Fluent. --]] local Fluent = _G.Fluent or error("Fluent library not found! Make sure Fluent is loaded.") local VantaLibrary = {} VantaLibrary.__index = VantaLibrary function VantaLibrary.new() local self = setmetatable({}, VantaLibrary) self.Window = nil self.Tabs = {} self.Options = Fluent.Options return self end -- Create main window function VantaLibrary:CreateWindow(title, subtitle, size, acrylic, theme, minimizeKey) self.Window = Fluent:CreateWindow({ Title = title or "Vanta UI", SubTitle = subtitle or "", Size = size or UDim2.fromOffset(580, 460), Acrylic = acrylic ~= false, Theme = theme or "Dark", MinimizeKey = minimizeKey or Enum.KeyCode.LeftControl }) return self.Window end -- Add a new tab function VantaLibrary:AddTab(name, icon) assert(self.Window, "Window must be created first") local tab = self.Window:AddTab({Title = name, Icon = icon or ""}) self.Tabs[name] = tab return tab end -- Button function VantaLibrary:AddButton(tab, title, description, callback) return tab:AddButton({Title = title, Description = description, Callback = callback}) end -- Toggle function VantaLibrary:AddToggle(tab, key, title, default, onChanged) local toggle = tab:AddToggle(key, {Title = title, Default = default}) if onChanged then toggle:OnChanged(function() onChanged(self.Options[key].Value) end) end return toggle end -- Slider function VantaLibrary:AddSlider(tab, key, title, description, default, min, max, rounding, callback) local slider = tab:AddSlider(key, { Title = title, Description = description, Default = default, Min = min, Max = max, Rounding = rounding, Callback = callback }) return slider end -- Dropdown / Multi-dropdown function VantaLibrary:AddDropdown(tab, key, title, values, multi, default, onChanged) local dropdown = tab:AddDropdown(key, {Title = title, Values = values, Multi = multi, Default = default}) if onChanged then dropdown:OnChanged(onChanged) end return dropdown end -- Colorpicker function VantaLibrary:AddColorpicker(tab, key, title, default, transparency, onChanged) local cp = tab:AddColorpicker(key, {Title = title, Default = default or Color3.fromRGB(255,255,255), Transparency = transparency or nil}) if onChanged then cp:OnChanged(onChanged) end return cp end -- Keybind function VantaLibrary:AddKeybind(tab, key, title, mode, default, callback, changedCallback) local kb = tab:AddKeybind(key, {Title = title, Mode = mode or "Toggle", Default = default or "LeftControl", Callback = callback, ChangedCallback = changedCallback}) return kb end -- Input function VantaLibrary:AddInput(tab, key, title, default, placeholder, numeric, finished, callback) local input = tab:AddInput(key, {Title = title, Default = default, Placeholder = placeholder, Numeric = numeric or false, Finished = finished or false, Callback = callback}) return input end -- Notification function VantaLibrary:Notify(title, content, subcontent, duration) Fluent:Notify({Title = title, Content = content, SubContent = subcontent, Duration = duration}) end -- Dialog function VantaLibrary:Dialog(title, content, buttons) assert(self.Window, "Window must be created first") self.Window:Dialog({Title = title, Content = content, Buttons = buttons}) end return VantaLibrary
